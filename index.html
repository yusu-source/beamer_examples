Титульный слайд titlepage 
% В преамбуле:
\title{Название доклада}
\subtitle{Подзаголовок (опционально)}
\author{Имя Фамилия}
\institute{Университет / Компания}
\date{\today}

% В document:
\begin{frame}
  \titlepage
\end{frame}

% Или без нумерации:
{
  \setbeamertemplate{footline}{}
  \setbeamertemplate{headline}{}
  \frame{\titlepage}
  \addtocounter{framenumber}{-1}
}

Листинг кода lstlisting / fragile 
% В преамбуле:
\usepackage{listings}
\lstset{
  basicstyle=\ttfamily\small,
  keywordstyle=\bfseries,
  commentstyle=\itshape\color{gray},
  numbers=left,
  numberstyle=\tiny,
  frame=single,
  breaklines=true
}

% На слайде (обязательно [fragile]):
\begin{frame}[fragile]{Пример кода}
  \begin{lstlisting}[language=Python]
def hello(name):
    print(f"Hello, {name}!")
    return True
  \end{lstlisting}
\end{frame}

Формулы equation / align 
\begin{frame}{Основная формула}
  Главное уравнение:
  \begin{equation}
    E = mc^2
  \end{equation}

  Система уравнений:
  \begin{align}
    a_1 x + b_1 y &= c_1 \\
    a_2 x + b_2 y &= c_2
  \end{align}

  Инлайн: $f(x) = \sum_{i=1}^{n} x_i$
\end{frame}

  \begin{frame}
    \vfill
    \centering

      shadow=true,rounded=true]{title}
      \usebeamerfont{title}\insertsectionhead\par
    \end{beamercolorbox}
    \vfill
  \end{frame}
}

\begin{tikzpicture}[
  node distance=7mm,
  box/.style={draw,rounded corners,minimum width=19mm,
              minimum height=8mm,align=center},
  >={Latex}
]
  \node[box] (login) {login};
  \node[box,right=of login] (pam) {PAM};
  \node[box,right=of pam] (db) {/etc/shadow};
  \node[box,right=of db] (session) {Session};
  \draw[->] (login) -- (pam);
  \draw[->] (pam) -- (db);
  \draw[->] (db) -- (session);
